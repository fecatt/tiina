import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {decode,encode,component} from '../src/codec.ts';
import {nativeGenerator,setNativeGenerator} from '../src/native-generator.ts';

const sineUrl=new URL('../reports/oracle-cases/RC_native_sine.TSC',import.meta.url);
const cfgUrl=new URL('../reports/oracle-cases/RC_native_configured.TSC',import.meta.url);

function load(url:URL):Uint8Array{
 return new Uint8Array(readFileSync(url));
}
function findGenerator(doc:any):{index:number;component:any}{
 for(let i=0;i<doc.records.length;i++){
  const c=component(doc.records[i],i);
  if(nativeGenerator(c))return {index:i,component:c};
 }
 throw Error('no native generator found');
}
function loadDoc(url:URL){return decode(load(url));}
function diffMask(a:Uint8Array,b:Uint8Array,allowed:(i:number)=>boolean){
 assert.equal(a.length,b.length,'length changed');
 const changed:number[]=[];
 for(let i=0;i<a.length;i++)if(a[i]!==b[i]){changed.push(i);if(!allowed(i))throw Error(`byte ${i} changed unexpectedly: ${a[i]}->${b[i]}`);}
 return changed;
}

// --- Read sine fixture and verify parsed values
{
 const doc=loadDoc(sineUrl);
 const {component:c}=findGenerator(doc);
 const s=nativeGenerator(c);
 assert.ok(s,'native sine parsed');
 assert.equal(s!.waveform,'sine');
 if(s!.waveform==='sine'){
  assert.ok(Number.isFinite(s!.offset));
  assert.ok(Number.isFinite(s!.amplitude)&&s!.amplitude>=0);
  assert.ok(Number.isFinite(s!.frequency)&&s!.frequency>0);
  assert.ok(Number.isFinite(s!.phase));
 }
}

// --- Read configured (rectangle) fixture and verify parsed values
{
 const doc=loadDoc(cfgUrl);
 const {component:c}=findGenerator(doc);
 const s=nativeGenerator(c);
 assert.ok(s);
 assert.equal(s!.waveform,'rectangle');
 if(s!.waveform==='rectangle'){
  assert.ok(s!.edge>0);
  assert.ok(s!.edge<1/s!.frequency/4);
 }
}

// --- Round-trip encode/decode of sine fixture preserves bytes
{
 const bytes=load(sineUrl);
 const doc=decode(bytes);
 const again=encode(doc);
 assert.deepEqual(Array.from(again),Array.from(bytes));
}

// --- Mutation: change sine parameters, verify only allowed bytes change
{
 const doc=loadDoc(sineUrl);
 const {index,c}=findGenerator(doc);
 const p=c.parameter_offset;
 const before=Uint8Array.from(doc.records[index].data);
 const orig=nativeGenerator(c)!;
 assert.equal(orig.waveform,'sine');
 const newOffset=orig.offset+1.5;
 const newAmplitude=orig.amplitude*0.5+0.25;
 const newFrequency=orig.frequency*2;
 const newPhase=45;
 setNativeGenerator(doc,index,{waveform:'sine',offset:newOffset,amplitude:newAmplitude,frequency:newFrequency,phase:newPhase});
 const after=doc.records[index].data;
 const changed=diffMask(before,after,i=>{
  return (i>=p&&i<p+8)||i===p+9||(i>=p+10&&i<p+34);
 });
 // verify floats
 const v=new DataView(after.buffer,after.byteOffset,after.byteLength);
 assert.equal(v.getFloat64(p,true),newOffset);
 assert.equal(v.getFloat64(p+10,true),newAmplitude);
 assert.equal(v.getFloat64(p+18,true),newFrequency);
 assert.equal(v.getFloat64(p+26,true),newPhase-90);
 assert.equal(after[p+9],4);
 // reparse
 const c2=component(doc.records[index],index);
 const s2=nativeGenerator(c2)!;
 assert.equal(s2.waveform,'sine');
 if(s2.waveform==='sine'){
  assert.equal(s2.offset,newOffset);
  assert.equal(s2.amplitude,newAmplitude);
  assert.equal(s2.frequency,newFrequency);
  assert.equal(s2.phase,newPhase);
 }
 // trailer & other records must remain byte-identical (covered by diffMask on trailer; whole-file check)
 const originalBytes=load(sineUrl);
 const reencoded=encode(doc);
 for(let i=0;i<originalBytes.length;i++){
  if(i<doc.records[index].data_offset||i>=doc.records[index].data_offset+before.length)continue;
 }
 // Explicit trailer check: bytes outside this record must be unchanged
 {
  const dataOffset=c.record.data_offset;
  for(let i=0;i<originalBytes.length;i++){
   if(i>=dataOffset&&i<dataOffset+before.length)continue;
   assert.equal(reencoded[i],originalBytes[i],`non-record byte ${i} changed`);
  }
 }
}

// --- Mutation: change rectangle parameters
{
 const doc=loadDoc(cfgUrl);
 const {index,c}=findGenerator(doc);
 const p=c.parameter_offset;
 const before=Uint8Array.from(doc.records[index].data);
 const orig=nativeGenerator(c)!;
 assert.equal(orig.waveform,'rectangle');
 if(orig.waveform!=='rectangle')throw Error('expected rectangle');
 const newEdge=orig.edge*0.5+0.001;
 setNativeGenerator(doc,index,{waveform:'rectangle',offset:orig.offset+2,amplitude:orig.amplitude+0.1,frequency:orig.frequency,edge:newEdge});
 const after=doc.records[index].data;
 diffMask(before,after,i=>(i>=p&&i<p+8)||i===p+9||(i>=p+10&&i<p+34));
 assert.equal(after[p+9],5);
 const v=new DataView(after.buffer,after.byteOffset,after.byteLength);
 assert.equal(v.getFloat64(p+26,true),newEdge);
}

// --- Invalid numeric values reject with byte-identical rollback
const invalidCases:[string,any][]=[
 ['negative amplitude',(s:any)=>({...s,amplitude:-1})],
 ['zero frequency',(s:any)=>({...s,frequency:0})],
 ['NaN offset',(s:any)=>({...s,offset:NaN})],
 ['Infinity frequency',(s:any)=>({...s,frequency:Infinity})],
];
for(const [name,mut] of invalidCases){
 const doc=loadDoc(sineUrl);
 const {index,c}=findGenerator(doc);
 const orig=nativeGenerator(c)!;
 const before=Uint8Array.from(doc.records[index].data);
 let threw=false;
 try{setNativeGenerator(doc,index,mut(orig));}catch{threw=true;}
 assert.ok(threw,`${name} should reject`);
 const after=doc.records[index].data;
 assert.deepEqual(Array.from(after),Array.from(before),`${name}: rollback byte-identical`);
}

// Rectangle edge invalid (>= quarter period)
{
 const doc=loadDoc(cfgUrl);
 const {index,c}=findGenerator(doc);
 const orig=nativeGenerator(c)!;
 if(orig.waveform!=='rectangle')throw Error('expected rectangle');
 const before=Uint8Array.from(doc.records[index].data);
 let threw=false;
 try{setNativeGenerator(doc,index,{waveform:'rectangle',offset:orig.offset,amplitude:orig.amplitude,frequency:orig.frequency,edge:1/orig.frequency});}catch{threw=true;}
 assert.ok(threw);
 assert.deepEqual(Array.from(doc.records[index].data),Array.from(before));
}

// --- Foreign waveform marker rejects
{
 const doc=loadDoc(sineUrl);
 const {index,c}=findGenerator(doc);
 const rec=doc.records[index];
 const orig=rec.data;
 const bad=Uint8Array.from(orig);
 bad[c.parameter_offset+9]=7;
 rec.data=bad;
 assert.equal(nativeGenerator(component(rec,index)),null,'foreign waveform marker rejected');
 rec.data=orig;
}

// --- Nonzero resistance rejects
{
 const doc=loadDoc(sineUrl);
 const {index,c}=findGenerator(doc);
 const rec=doc.records[index];
 const orig=rec.data;
 const bad=Uint8Array.from(orig);
 const v=new DataView(bad.buffer,bad.byteOffset,bad.byteLength);
 v.setFloat64(c.parameter_offset+34,1.0,true);
 rec.data=bad;
 assert.equal(nativeGenerator(component(rec,index)),null,'nonzero resistance rejected');
 rec.data=orig;
}

// --- Nonzero backing-buffer offset fixture: exercise DataView.byteOffset path
{
 const doc=loadDoc(sineUrl);
 const {index,c}=findGenerator(doc);
 const dataLen=c.record.data.length+64;
 const backing=new Uint8Array(dataLen);
 const offset=32;
 backing.set(c.record.data,offset);
 // place data in a subarray whose byteOffset is nonzero
 const sub=backing.subarray(offset,offset+c.record.data.length);
 const rec={...c.record,data:sub};
 const comp=component(rec,index);
 const s=nativeGenerator(comp);
 assert.ok(s,'parses with nonzero byteOffset');
 // mutate via setNativeGenerator on a doc containing this record
 const doc2:any={records:[rec],...doc};
 // need component lookup; setNativeGenerator uses doc.records[index]
 const before=Uint8Array.from(backing);
 const target=nativeGenerator(comp)!;
 const newOffset=target.offset+3;
 setNativeGenerator(doc2,0,{waveform:'sine',offset:newOffset,amplitude:target.amplitude,frequency:target.frequency,phase:target.waveform==='sine'?target.phase+10:10});
 // verify mutation landed at correct physical bytes in backing buffer
 const v2=new DataView(backing.buffer,backing.byteOffset,backing.byteLength);
 assert.equal(v2.getFloat64(offset+c.parameter_offset,true),newOffset);
 // and reparse
 const s3=nativeGenerator(component(doc2.records[0],0))!;
 assert.equal(s3.offset,newOffset);
}

// --- Whole-file encode after sine mutation: unknown bytes/records/trailer exact
{
 const originalBytes=load(sineUrl);
 const doc=decode(originalBytes);
 const {index,c}=findGenerator(doc);
 const p=c.parameter_offset;
 const orig=nativeGenerator(c)!;
 setNativeGenerator(doc,index,{waveform:'sine',offset:orig.offset+5,amplitude:orig.amplitude,frequency:orig.frequency,phase:orig.waveform==='sine'?orig.phase:0});
 const re=encode(doc);
 assert.equal(re.length,originalBytes.length);
 const dataOffset=c.record.data_offset;
 const dataLength=c.record.data.length;
 for(let i=0;i<originalBytes.length;i++){
  if(i>=dataOffset&&i<dataOffset+dataLength){
   const rel=i-dataOffset;
   const allowed=(rel>=p&&rel<p+8)||rel===p+9||(rel>=p+10&&rel<p+34);
   if(allowed)continue;
  }
  assert.equal(re[i],originalBytes[i],`byte ${i} outside allowed region changed`);
 }
}
